# Python Program 1: Smart City Traffic Prediction System
# Simulates traffic flow prediction using basic ML concepts

import random
from datetime import datetime, timedelta

class SmartCityTrafficPredictor:
    def __init__(self):
        self.historical_data = []
        self.traffic_patterns = {
            'morning_rush': (7, 9, 0.8),  # hour_start, hour_end, congestion_level
            'evening_rush': (17, 19, 0.85),
            'midday': (12, 14, 0.5),
            'night': (22, 6, 0.2)
        }
    
    def generate_traffic_data(self, days=30):
        """Simulate historical traffic data"""
        start_date = datetime.now() - timedelta(days=days)
        
        for day in range(days):
            current_date = start_date + timedelta(days=day)
            for hour in range(24):
                congestion = self.calculate_congestion(hour, current_date.weekday())
                self.historical_data.append({
                    'datetime': current_date.replace(hour=hour),
                    'hour': hour,
                    'day_of_week': current_date.weekday(),
                    'congestion_level': congestion
                })
    
    def calculate_congestion(self, hour, day_of_week):
        """Calculate congestion based on time patterns"""
        base_congestion = 0.3
        
        # Weekend reduction
        if day_of_week >= 5:
            base_congestion *= 0.6
        
        # Time-based patterns
        for pattern, (start, end, level) in self.traffic_patterns.items():
            if start <= hour < end or (start > end and (hour >= start or hour < end)):
                base_congestion = max(base_congestion, level)
        
        # Add some randomness
        return min(1.0, base_congestion + random.uniform(-0.1, 0.1))
    
    def predict_traffic(self, target_hour, target_day):
        """Simple prediction based on historical averages"""
        relevant_data = [
            d['congestion_level'] 
            for d in self.historical_data 
            if d['hour'] == target_hour and d['day_of_week'] == target_day
        ]
        
        if relevant_data:
            prediction = sum(relevant_data) / len(relevant_data)
            return prediction
        return 0.5  # Default prediction
    
    def get_traffic_recommendation(self, congestion_level):
        """Provide recommendations based on predicted congestion"""
        if congestion_level < 0.3:
            return "Low traffic - Normal routes recommended"
        elif congestion_level < 0.6:
            return "Moderate traffic - Consider alternative routes"
        elif congestion_level < 0.8:
            return "Heavy traffic - Alternative routes strongly recommended"
        else:
            return "Severe congestion - Use public transportation if possible"

# Example usage
if __name__ == "__main__":
    predictor = SmartCityTrafficPredictor()
    predictor.generate_traffic_data(days=60)
    
    print("=== Smart City Traffic Prediction System ===\n")
    
    # Predict traffic for next week
    test_day = datetime.now() + timedelta(days=1)
    
    for hour in [8, 12, 18, 22]:  # Key hours
        prediction = predictor.predict_traffic(hour, test_day.weekday())
        recommendation = predictor.get_traffic_recommendation(prediction)
        
        print(f"Time: {hour}:00")
        print(f"Predicted Congestion: {prediction:.2%}")
        print(f"Recommendation: {recommendation}")
        print("-" * 50)

# Python Program 2: V2X Communication Simulator
# Simulates vehicle-to-vehicle communication for collision avoidance

import math
import random

class Vehicle:
    def __init__(self, vehicle_id, x, y, speed, direction):
        self.id = vehicle_id
        self.x = x
        self.y = y
        self.speed = speed  # km/h
        self.direction = direction  # degrees (0 = North, 90 = East)
        self.communication_range = 300  # meters
        
    def update_position(self, time_delta):
        """Update vehicle position based on speed and direction"""
        # Convert speed to m/s and calculate displacement
        speed_ms = self.speed / 3.6
        distance = speed_ms * time_delta
        
        # Calculate new position
        rad_direction = math.radians(self.direction)
        self.x += distance * math.sin(rad_direction)
        self.y += distance * math.cos(rad_direction)
    
    def distance_to(self, other_vehicle):
        """Calculate distance to another vehicle"""
        return math.sqrt((self.x - other_vehicle.x)**2 + 
                        (self.y - other_vehicle.y)**2)
    
    def can_communicate(self, other_vehicle):
        """Check if vehicle is within communication range"""
        return self.distance_to(other_vehicle) <= self.communication_range
    
    def predict_collision_risk(self, other_vehicle, time_horizon=5):
        """Predict collision risk with another vehicle"""
        # Simple prediction: extrapolate positions
        my_future_x = self.x + (self.speed / 3.6) * time_horizon * math.sin(math.radians(self.direction))
        my_future_y = self.y + (self.speed / 3.6) * time_horizon * math.cos(math.radians(self.direction))
        
        other_future_x = other_vehicle.x + (other_vehicle.speed / 3.6) * time_horizon * math.sin(math.radians(other_vehicle.direction))
        other_future_y = other_vehicle.y + (other_vehicle.speed / 3.6) * time_horizon * math.cos(math.radians(other_vehicle.direction))
        
        future_distance = math.sqrt((my_future_x - other_future_x)**2 + 
                                   (my_future_y - other_future_y)**2)
        
        # Critical distance threshold (5 meters)
        if future_distance < 5:
            return "HIGH RISK"
        elif future_distance < 15:
            return "MEDIUM RISK"
        else:
            return "LOW RISK"

class V2XSimulator:
    def __init__(self):
        self.vehicles = []
        self.message_count = 0
        self.collision_warnings = 0
        
    def add_vehicle(self, vehicle):
        self.vehicles.append(vehicle)
    
    def simulate_step(self, time_delta=1.0):
        """Simulate one time step"""
        # Update all vehicle positions
        for vehicle in self.vehicles:
            vehicle.update_position(time_delta)
        
        # Check V2V communications and collision risks
        for i, vehicle1 in enumerate(self.vehicles):
            for vehicle2 in self.vehicles[i+1:]:
                if vehicle1.can_communicate(vehicle2):
                    self.message_count += 1
                    
                    # Check collision risk
                    risk = vehicle1.predict_collision_risk(vehicle2)
                    if risk in ["HIGH RISK", "MEDIUM RISK"]:
                        self.collision_warnings += 1
                        print(f"⚠️ COLLISION WARNING: Vehicle {vehicle1.id} and {vehicle2.id}")
                        print(f"   Risk Level: {risk}")
                        print(f"   Distance: {vehicle1.distance_to(vehicle2):.1f}m")
    
    def generate_report(self):
        """Generate simulation statistics"""
        print("\n" + "="*60)
        print("V2X Communication Simulation Report")
        print("="*60)
        print(f"Total Vehicles: {len(self.vehicles)}")
        print(f"V2V Messages Exchanged: {self.message_count}")
        print(f"Collision Warnings Issued: {self.collision_warnings}")
        print(f"Average Messages per Vehicle: {self.message_count / len(self.vehicles):.1f}")

# Example usage
if __name__ == "__main__":
    print("=== V2X Communication Simulator ===\n")
    
    simulator = V2XSimulator()
    
    # Create vehicles with different trajectories
    simulator.add_vehicle(Vehicle("V001", 0, 0, 60, 0))      # North
    simulator.add_vehicle(Vehicle("V002", 100, 0, 50, 0))    # North (behind)
    simulator.add_vehicle(Vehicle("V003", 0, -200, 70, 0))   # North (approaching)
    simulator.add_vehicle(Vehicle("V004", 50, -100, 55, 90)) # East (crossing)
    simulator.add_vehicle(Vehicle("V005", -50, -150, 65, 45)) # Northeast
    
    # Run simulation for 10 seconds
    print("Running 10-second simulation...\n")
    for t in range(10):
        print(f"Time: {t}s")
        simulator.simulate_step(1.0)
        print("-" * 60)
    
    simulator.generate_report()

# Python Program 3: Telecom Market Analysis Tool
# Analyzes market trends and competitive positioning

import random
from datetime import datetime

class TelecomCompany:
    def __init__(self, name, market_share, ai_investment, edge_capability):
        self.name = name
        self.market_share = market_share  # percentage
        self.ai_investment = ai_investment  # millions USD
        self.edge_capability = edge_capability  # 1-10 scale
        self.revenue = 0
        self.growth_rate = 0
        
    def calculate_competitive_score(self):
        """Calculate overall competitive score"""
        # Weighted scoring: market share (40%), AI investment (30%), edge capability (30%)
        score = (self.market_share * 0.4 + 
                (self.ai_investment / 1000) * 0.3 + 
                self.edge_capability * 3 * 0.3)
        return min(100, score)
    
    def project_growth(self, years=5):
        """Project market share growth based on investments"""
        # Simple model: higher AI investment and edge capability = higher growth
        base_growth = 2.0  # Base growth rate %
        ai_factor = (self.ai_investment / 500) * 1.5
        edge_factor = (self.edge_capability / 10) * 2.0
        
        self.growth_rate = base_growth + ai_factor + edge_factor
        future_share = self.market_share * ((1 + self.growth_rate/100) ** years)
        
        return min(100, future_share)  # Cap at 100%

class MarketAnalyzer:
    def __init__(self):
        self.companies = []
        self.market_size = 250000  # Million USD
        self.growth_rate = 8.5  # Annual % growth
        
    def add_company(self, company):
        self.companies.append(company)
    
    def normalize_market_shares(self):
        """Ensure market shares sum to 100%"""
        total_share = sum(c.market_share for c in self.companies)
        if total_share > 0:
            for company in self.companies:
                company.market_share = (company.market_share / total_share) * 100
    
    def calculate_revenues(self):
        """Calculate revenue for each company"""
        for company in self.companies:
            company.revenue = (company.market_share / 100) * self.market_size
    
    def competitive_analysis(self):
        """Perform competitive landscape analysis"""
        self.normalize_market_shares()
        self.calculate_revenues()
        
        print("\n" + "="*80)
        print("TELECOM MARKET COMPETITIVE ANALYSIS")
        print("="*80)
        print(f"Total Market Size: ${self.market_size:,}M")
        print(f"Market Growth Rate: {self.growth_rate}%")
        print("-"*80)
        
        # Sort by competitive score
        sorted_companies = sorted(self.companies, 
                                 key=lambda x: x.calculate_competitive_score(), 
                                 reverse=True)
        
        print(f"\n{'Company':<20} {'Market Share':<15} {'Revenue ($M)':<15} {'Comp. Score':<15}")
        print("-"*80)
        
        for company in sorted_companies:
            score = company.calculate_competitive_score()
            print(f"{company.name:<20} {company.market_share:>6.2f}% {company.revenue:>13,.0f} {score:>12.1f}")
        
        # Identify market leader and challengers
        leader = sorted_companies[0]
        print(f"\n📊 Market Leader: {leader.name} with {leader.market_share:.1f}% market share")
        
        # Calculate market concentration (HHI - Herfindahl-Hirschman Index)
        hhi = sum((c.market_share) ** 2 for c in self.companies)
        print(f"📈 Market Concentration (HHI): {hhi:.0f}", end=" - ")
        
        if hhi < 1500:
            print("Competitive Market")
        elif hhi < 2500:
            print("Moderately Concentrated")
        else:
            print("Highly Concentrated")
    
    def investment_analysis(self):
        """Analyze AI investment levels"""
        print("\n" + "="*80)
        print("AI INVESTMENT ANALYSIS")
        print("="*80)
        
        sorted_by_investment = sorted(self.companies, 
                                     key=lambda x: x.ai_investment, 
                                     reverse=True)
        
        total_investment = sum(c.ai_investment for c in self.companies)
        
        print(f"\n{'Company':<20} {'AI Investment ($M)':<20} {'% of Total':<15} {'Edge Score':<15}")
        print("-"*80)
        
        for company in sorted_by_investment:
            pct_total = (company.ai_investment / total_investment) * 100
            print(f"{company.name:<20} ${company.ai_investment:>10,.0f} {pct_total:>12.1f}% {company.edge_capability:>13}/10")
        
        print(f"\nTotal Industry AI Investment: ${total_investment:,.0f}M")
